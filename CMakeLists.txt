cmake_minimum_required(VERSION 3.24...4.0)
cmake_policy(SET CMP0091 NEW)
project(attempt_engine LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ATTEMPT_ENGINE_BUILD_BARD_BATTLE "Build Bard Battle" ON)
option(BARD_BATTLE_BUILD_CLIENT "Build the SDL/Vulkan Bard Battle client" ON)
option(BARD_BATTLE_BUILD_SERVER "Build the Bard Battle server and bot" ON)
include(CTest)

if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

function(attempt_engine_configure_target target_name)
    if(MSVC)
        target_compile_definitions(${target_name} PRIVATE NOMINMAX)
        target_compile_options(${target_name} PRIVATE /W4)
    else()
        target_compile_options(${target_name} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()

if(ATTEMPT_ENGINE_BUILD_BARD_BATTLE)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/external/enet/CMakeLists.txt")
        message(FATAL_ERROR "ENet source is missing. Run the Bard Battle platform build script.")
    endif()
    set(ENET_BUILD_SHARED OFF CACHE BOOL "" FORCE)
    set(ENET_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    add_subdirectory(external/enet EXCLUDE_FROM_ALL)

    if(BARD_BATTLE_BUILD_CLIENT)
        set(SDL_SHARED OFF CACHE BOOL "" FORCE)
        set(SDL_STATIC ON CACHE BOOL "" FORCE)
        set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
        set(SDL_TESTS OFF CACHE BOOL "" FORCE)
        if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/external/SDL/CMakeLists.txt")
            message(FATAL_ERROR "SDL source is missing. Run the Bard Battle Windows build script or initialize the SDL submodule.")
        endif()
        add_subdirectory(external/SDL)
        find_package(Vulkan REQUIRED)
    endif()
endif()

add_subdirectory(attempt_engine)
if(ATTEMPT_ENGINE_BUILD_BARD_BATTLE)
    add_subdirectory(games/bard_battle)
endif()
